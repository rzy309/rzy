二、for 循环 — 数7游戏（1~30）
// 数7规则：7的倍数 或 末尾是7 → 踩雷 var rzy_result = []; for (var i = 1; i <= 30; i++) { if (i % 7 === 0 || i % 10 === 7) { rzy_result.push('💥'); // 踩雷 } else { rzy_result.push(i); // 正常输出数字 } } console.log(rzy_result);

三、while / do-while — 猜数字（模拟）
// while ：累加直到和 >= 100 var rzy_sum = 0, rzy_n = 1; while (rzy_sum < 100) { rzy_sum += rzy_n; rzy_n++; } // do-while ：至少执行一次后再判断 do { // 提示输入 } while (rzy_input !== 'quit');

四、for...of（遍历数组）/ for...in（遍历对象键）
// for...of ：遍历数组元素 var rzy_colors = ['红', '橙', '黄', '绿', '蓝']; for (var color of rzy_colors) { console.log(color); } // for...in ：遍历对象属性 var rzy_student = { name: '荣子祎', age: 20, major: '前端' }; for (var key in rzy_student) { console.log(key + ' = ' + rzy_student[key]); }

// continue ：跳过偶数，只打印 1~20 内的奇数 for (var i = 1; i <= 20; i++) { if (i % 2 === 0) continue; // 跳过本次 console.log(i); } // break ：找到第一个 > 100 且是 7 的倍数就停止 for (var i = 101; ; i++) { if (i % 7 === 0) { console.log('找到：' + i); break; // 立即跳出 } }
